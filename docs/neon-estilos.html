/* <canvas id="rain"> dentro de .bg */
const c = document.getElementById('rain'), x = c.getContext('2d');
const chars = 'アカサタナハマヤラワ0123456789ｦｱｳｴｵｶｷｸｹｺ';
let cols, drops;
function size() {
  c.width = c.offsetWidth; c.height = c.offsetHeight;
  cols = Math.floor(c.width / 16); drops = Array(cols).fill(0).map(() => Math.random() * -50);
}
function tick() {
  x.fillStyle = 'rgba(1,8,5,.12)'; x.fillRect(0, 0, c.width, c.height);
  x.font = '16px "Share Tech Mono", monospace';
  drops.forEach((y, i) => {
    x.fillStyle = Math.random() > .96 ? '#eafff3' : '#00ff88';
    x.fillText(chars[Math.random() * chars.length | 0], i * 16, y * 16);
    drops[i] = y * 16 > c.height && Math.random() > .975 ? 0 : y + 1;
  });
}
size(); addEventListener('resize', size); setInterval(tick, 55);
